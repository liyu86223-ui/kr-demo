function performSynthesis(iId, rId, sId, nId, text) {

const i = document.getElementById(iId); const r = document.getElementById(rId);

const s = document.getElementById(sId); const n = document.getElementById(nId);

i.classList.add('animate-synthesis-out'); s.style.display = 'none';

speak(text); 

setTimeout(() => { 

i.style.display = 'none'; 

r.style.display = 'flex'; 

r.classList.add('fade-in-up'); 

n.style.display = 'block'; 

}, 450);

}

 

function handleMic(num) {

const btn = document.getElementById(`s${num}-mic`); const h = document.getElementById(`s${num}-hint`);

const t = document.getElementById(`s${num}-text`); const w = document.getElementById(`s${num}-wave`); const n = document.getElementById(`s${num}-next-btn`);

const charCount = t.innerText.replace(/\s+/g, '').length; 

const duration = 800 + (charCount * 150); 

btn.classList.add('recording'); w.classList.add('active'); 

btn.childNodes[0].textContent = ""; 

h.innerText = "正在倾听..."; h.style.color = "var(--recording-color)";

setTimeout(() => { 

btn.classList.remove('recording'); w.classList.remove('active'); 

btn.classList.add('done'); btn.innerHTML = "✅"; 

t.classList.add('success'); h.innerText = "读对了！"; 

h.style.color = "var(--success-color)"; n.style.display = "block"; 

}, duration); 

}
